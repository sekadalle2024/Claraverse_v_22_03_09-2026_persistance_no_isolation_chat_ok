const keys = Object.keys(localStorage).filter(k => k.includes('flowise'));
console.log('2. Clés Flowise:', keys.length);
let tableCount = 0;
keys.forEach(k => {
  const msgs = JSON.parse(localStorage.getItem(k));
  msgs.forEach((m, i) => {
    if (m.message && m.message.includes('<table')) {
      tableCount++;
      console.log(`✅ Table dans ${k} message ${i}`);
    }
  });
});
console.log('3. Total tables:', tableCount);
